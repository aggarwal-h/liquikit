## Usage

```tsx

import { Bold, Link } from "lucide-react"
```

```tsx

  <Toolbar.Button icon aria-label="Bold"><Bold className="size-4" /></Toolbar.Button>
  <Toolbar.Separator />
  <Toolbar.Button icon aria-label="Link"><Link className="size-4" /></Toolbar.Button>
</Toolbar.Root>
```
