## Usage

```tsx
"use client"

import { NavigationMenu } from "@/components/liquikit/navigation-menu"
```

```tsx
<NavigationMenu.Root>
  <NavigationMenu.List aria-label="Main">
    <NavigationMenu.Item>
      <NavigationMenu.Trigger>Products</NavigationMenu.Trigger>
      <NavigationMenu.Content>
        <NavigationMenu.Link href="/laptops" title="Laptops" />
      </NavigationMenu.Content>
    </NavigationMenu.Item>
  </NavigationMenu.List>
</NavigationMenu.Root>
```
