## Usage

```tsx
import { Tabs } from "@/components/liquikit/tabs"
```

```tsx
<Tabs.Root defaultValue="overview">
  <Tabs.List tabs={[{ value: "overview", label: "Overview" }, { value: "activity", label: "Activity" }]} />
  <Tabs.Panel value="overview">…</Tabs.Panel>
  <Tabs.Panel value="activity">…</Tabs.Panel>
</Tabs.Root>
```
