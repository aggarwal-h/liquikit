## Usage

```tsx
"use client"

import { Combobox } from "@/components/liquikit/combobox"
```

```tsx
<Combobox.Root items={fruits}>
  <Combobox.Input placeholder="Choose a fruit" aria-label="Fruit" />
  <Combobox.Popup empty="No fruit found.">
    <Combobox.List>
      {(fruit: string) => (
        <Combobox.Item key={fruit} value={fruit}>
          {fruit}
        </Combobox.Item>
      )}
    </Combobox.List>
  </Combobox.Popup>
</Combobox.Root>
```

## Examples

### Multiple

Pass `multiple` to `Combobox.Root` and use `Combobox.Chips` in place of the input. Chosen items sit in the field as chips. Remove one with its cross, or with Backspace from the start of the field.
