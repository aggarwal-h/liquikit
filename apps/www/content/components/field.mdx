## Usage

```tsx
import { Field, Input } from "@/components/liquikit/field"
```

```tsx
<Field.Root>
  <Field.Label>Email</Field.Label>
  <Input type="email" placeholder="you@example.com" />
  <Field.Description>For your receipts.</Field.Description>
</Field.Root>
```
