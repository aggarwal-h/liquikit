## Usage

```tsx
import { ContextMenu } from "@/components/liquikit/context-menu"
import { Menu } from "@/components/liquikit/menu"
```

```tsx
<ContextMenu.Root>
  <ContextMenu.Trigger>Right-click here</ContextMenu.Trigger>
  <ContextMenu.Popup>
    <Menu.Item>Get Info</Menu.Item>
    <Menu.Item>Rename</Menu.Item>
  </ContextMenu.Popup>
</ContextMenu.Root>
```

The items are the ones from [Menu](/docs/components/menu).
