## Usage

```tsx
"use client"

import { Autocomplete } from "@/components/liquikit/autocomplete"
```

```tsx
<Autocomplete.Root items={cities}>
  <Autocomplete.Input placeholder="Search cities" aria-label="City" />
  <Autocomplete.Popup empty="No city found.">
    <Autocomplete.List>
      {(city: string) => (
        <Autocomplete.Item key={city} value={city}>
          {city}
        </Autocomplete.Item>
      )}
    </Autocomplete.List>
  </Autocomplete.Popup>
</Autocomplete.Root>
```

Use [Combobox](/docs/components/combobox) when the value must be one of the items.
