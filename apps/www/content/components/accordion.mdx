## Usage

```tsx
"use client"

import { Accordion } from "@/components/liquikit/accordion"
```

```tsx
<Accordion.Root defaultValue={["wifi"]}>
  <Accordion.Item value="wifi">
    <Accordion.Trigger>Wi-Fi</Accordion.Trigger>
    <Accordion.Panel>Connected to Home.</Accordion.Panel>
  </Accordion.Item>
</Accordion.Root>
```

Pass `multiple` to `Accordion.Root` to let more than one section stay open.
